// 3. Valheim Archive SDK Client
// ============================================================================

export interface ValheimArchiveClientConfig {
  baseUrl?: string;
  headers?: Record<string, string>;
  fetchFn?: typeof fetch;
}

export class ValheimArchiveClient {
  private baseUrl: string;
  private headers: Record<string, string>;
  private fetchFn: typeof fetch;

  constructor(config: ValheimArchiveClientConfig = {}) {
    this.baseUrl = (config.baseUrl || '').replace(/\/+$/, '');
    this.headers = {
      'Content-Type': 'application/json',
      Accept: 'application/json',
      ...(config.headers || {}),
    };
    this.fetchFn = config.fetchFn || (typeof window !== 'undefined' ? window.fetch.bind(window) : fetch);
  }

  private async request<T>(path: string, options: RequestInit = {}): Promise<T> {
    const url = `${this.baseUrl}${path.startsWith('/') ? path : `/${path}`}`;
    const res = await this.fetchFn(url, {
      ...options,
      headers: {
        ...this.headers,
        ...(options.headers || {}),
      },
    });

    if (!res.ok) {
      let errBody: unknown;
      try {
        errBody = await res.json();
      } catch {
        errBody = await res.text();
      }
      throw new Error(`ValheimArchiveClient Error [${res.status} ${res.statusText}] at ${path}: ${JSON.stringify(errBody)}`);
    }

    return (await res.json()) as T;
  }

  // --- Gateway & Archetypes ---
  public async getGatewayStats(): Promise<GatewayStatsView> {
    return this.request<GatewayStatsView>('/api/gateway/stats');
  }

  public async getArchetypes(): Promise<ArchetypeDescriptor[]> {
    return this.request<ArchetypeDescriptor[]>('/api/gateway/archetypes');
  }

  // --- Builders Directory ---
  public async listBuilders(params: BuilderDirectoryParams = {}): Promise<BuilderDirectoryResponse> {
    const search = new URLSearchParams();
    if (params.q) search.set('q', params.q);
    if (params.era !== undefined) search.set('era', String(params.era));
    if (params.withAlbums !== undefined) search.set('with_albums', String(params.withAlbums));
    if (params.sort) search.set('sort', params.sort);
    if (params.cursor) search.set('cursor', params.cursor);
    if (params.limit) search.set('limit', String(params.limit));

    const qs = search.toString();
    return this.request<BuilderDirectoryResponse>(`/api/creators${qs ? `?${qs}` : ''}`);
  }

  // --- Viking "MySpace" Profiles ---
  public async getBuilderProfile(builderId: string): Promise<BuilderProfileView> {
    return this.request<BuilderProfileView>(`/api/creators/${encodeURIComponent(builderId)}`);
  }

  public async getBuilderEraLedger(builderId: string, era?: number): Promise<EraBuildRecord[]> {
    const qs = era !== undefined ? `?era=${era}` : '';
    return this.request<EraBuildRecord[]>(`/api/creators/${encodeURIComponent(builderId)}/ledger${qs}`);
  }

  // --- Commands & Dispatch ---
  public async dispatchDiscordPhotoTicket(request: PhotoDispatchRequest): Promise<DiscordTicketResponse> {
    return this.request<DiscordTicketResponse>('/api/dispatch/discord-ticket', {
      method: 'POST',
      body: JSON.stringify(request),
    });
  }

  public async generateClientClaimPayload(claim: ClaimBuildRequest): Promise<ClientPayloadReceipt> {
    return this.request<ClientPayloadReceipt>('/api/claims/generate-payload', {
      method: 'POST',
      body: JSON.stringify(claim),
    });
  }

  public async inscribeStele(command: InscribeSteleRequest): Promise<RunicInscription> {
    return this.request<RunicInscription>('/api/v1/commands/stele/inscribe', {
      method: 'POST',
      body: JSON.stringify(command),
    });
  }

  public async customizeProfile(command: CustomizeProfileRequest): Promise<{ success: boolean; builderId: string }> {
    return this.request<{ success: boolean; builderId: string }>('/api/v1/commands/profile/customize', {
      method: 'POST',
      body: JSON.stringify(command),
    });
  }
}

// Export singleton instance for direct Next.js / frontend use
export const valheimArchive = new ValheimArchiveClient();

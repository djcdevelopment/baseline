"""
The Comfy Community — Valheim Chronicler Archive
FastAPI & NumPy Kinship Overlap Engine: Dynamic Shield-Wall Top 8 Calculation

Architecture: High-Performance Linear Algebra & Graph Projection Service
Formula:
    KinshipScore(A, B) = sum_{b in SharedBuilds} [ min(P_{A,b}, P_{B,b}) * log10(TotalPieces_b) ]

Features:
- Vectorized NumPy & SciPy Sparse Matrix computation across millions of raw pieces
- Automated role derivation (Forge Lead, Rampart Eng, Timber Partner, Masonry Ally)
- Dynamic Top-8 Shield-Wall extraction with key shared monument attribution
- Memory-efficient batch replay and Redis cache integration
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from typing import Dict, List, Optional, Tuple
import numpy as np
from scipy import sparse
from pydantic import BaseModel, Field

logger = logging.getLogger("valheim.kinship_engine")


# ============================================================================
# 1. Domain Models (Pydantic V2)
# ============================================================================

class KinshipCollaborator(BaseModel):
    """Dynamic Shield-Wall Top 8 collaborator calculated via attribution overlap."""
    collaborator_id: str = Field(..., description="Unique builder identifier")
    collaborator_name: str = Field(..., description="Display name / primary handle")
    role_tag: str = Field(..., description="Heuristic derived role (e.g. 'Forge Lead', 'Foundation Mason')")
    shared_builds_count: int = Field(..., ge=0, description="Total shared builds across all eras")
    shared_pieces_count: int = Field(..., ge=0, description="Total pieces placed across shared builds")
    kinship_score: float = Field(..., ge=0.0, description="Mathematically derived kinship overlap score")
    key_shared_build_id: str = Field(..., description="Build ID with the highest single collaborative impact")


class BuildContributionRecord(BaseModel):
    build_id: str
    era: int
    total_pieces: int
    builder_id: str
    builder_name: str
    pieces_placed: int


# ============================================================================
# 2. Kinship Engine (NumPy & SciPy Vectorized Pipeline)
# ============================================================================

class KinshipOverlapEngine:
    """
    High-throughput graph projection engine calculating multi-builder kinship matrices.
    Uses sparse CSR matrices for memory-efficient calculation over 2,500+ builders
    and 12,000+ build clusters spanning 9 wiped eras.
    """

    def __init__(self, min_shared_pieces: int = 10, log_scale_base: float = 10.0):
        self.min_shared_pieces = min_shared_pieces
        self.log_scale_base = log_scale_base
        
        # Builder & Build Index Mappings
        self.builder_id_to_idx: Dict[str, int] = {}
        self.idx_to_builder_id: Dict[int, str] = {}
        self.builder_names: Dict[str, str] = {}
        
        self.build_id_to_idx: Dict[str, int] = {}
        self.idx_to_build_id: Dict[int, str] = {}
        self.build_piece_counts: np.ndarray = np.array([], dtype=np.float64)

    def build_matrix_from_records(
        self, records: List[BuildContributionRecord]
    ) -> Tuple[sparse.csr_matrix, np.ndarray]:
        """
        Transforms flat relational / event records into a normalized builder-build sparse matrix:
        Rows = Builders (N)
        Cols = Builds (M)
        Values = Contribution Share Percentage P_{A, b} in [0.0, 1.0]
        """
        builder_set = set()
        build_dict: Dict[str, int] = {}  # build_id -> total_pieces

        for r in records:
            builder_set.add((r.builder_id, r.builder_name))
            if r.build_id not in build_dict:
                build_dict[r.build_id] = r.total_pieces

        # Build index maps
        self.builder_id_to_idx = {b_id: idx for idx, (b_id, _) in enumerate(sorted(builder_set))}
        self.idx_to_builder_id = {idx: b_id for b_id, idx in self.builder_id_to_idx.items()}
        self.builder_names = {b_id: name for b_id, name in builder_set}

        self.build_id_to_idx = {b_id: idx for idx, b_id in enumerate(sorted(build_dict.keys()))}
        self.idx_to_build_id = {idx: b_id for b_id, idx in self.build_id_to_idx.items()}

        num_builders = len(self.builder_id_to_idx)
        num_builds = len(self.build_id_to_idx)

        # Vectorized build piece weights: log10(TotalPieces_b)
        self.build_piece_counts = np.zeros(num_builds, dtype=np.float64)
        for b_id, total_pcs in build_dict.items():
            b_idx = self.build_id_to_idx[b_id]
            # Safe log scaling: ensure >= 1.0 piece
            self.build_piece_counts[b_idx] = np.log10(max(float(total_pcs), 1.0))

        # Build COO sparse matrix
        rows = []
        cols = []
        shares = []

        for r in records:
            b_idx = self.builder_id_to_idx[r.builder_id]
            build_idx = self.build_id_to_idx[r.build_id]
            total_pcs = max(build_dict[r.build_id], 1)
            share_pct = float(r.pieces_placed) / float(total_pcs)

            rows.append(b_idx)
            cols.append(build_idx)
            shares.append(share_pct)

        matrix = sparse.coo_matrix(
            (shares, (rows, cols)), shape=(num_builders, num_builds), dtype=np.float32
        ).tocsr()

        return matrix, self.build_piece_counts

    def compute_kinship_for_builder(
        self,
        target_builder_id: str,
        contribution_matrix: sparse.csr_matrix,
        build_weights: np.ndarray,
        records: List[BuildContributionRecord],
        top_k: int = 8
    ) -> List[KinshipCollaborator]:
        """
        Computes the exact KinshipScore(A, B) for target builder A against all co-builders.
        
        Formula:
            KinshipScore(A, B) = sum_{b} [ min(P_{A,b}, P_{B,b}) * log10(TotalPieces_b) ]
        """
        if target_builder_id not in self.builder_id_to_idx:
            logger.warning(f"Builder {target_builder_id} not indexed in kinship graph.")
            return []

        target_idx = self.builder_id_to_idx[target_builder_id]
        target_row = contribution_matrix.getrow(target_idx)
        
        # Non-zero builds for target builder
        shared_build_indices = target_row.indices
        if len(shared_build_indices) == 0:
            return []

        target_shares = target_row.data  # P_{target, b}
        weights_sub = build_weights[shared_build_indices]  # log10(TotalPieces_b)

        # Slice sub-matrix of all co-builders across these specific builds
        # Sub-matrix shape: (num_builders, num_target_builds)
        sub_matrix = contribution_matrix[:, shared_build_indices].tocsc()

        # Iterate only through candidate builders who share at least 1 build
        candidate_co_builders = np.unique(sub_matrix.tocoo().row)
        # Exclude target builder self
        candidate_co_builders = candidate_co_builders[candidate_co_builders != target_idx]

        if len(candidate_co_builders) == 0:
            return []

        # Vectorized min(P_A, P_B) * log10(Pieces) per candidate
        # Pre-index records for deep metadata (role, shared build counts, top monument)
        builder_shared_stats: Dict[int, Dict] = {
            c_idx: {
                "shared_builds": set(),
                "shared_pieces": 0,
                "best_build_id": "",
                "best_build_score": -1.0,
            }
            for c_idx in candidate_co_builders
        }

        # Filter relevant records for fast attribution lookup
        target_build_set = {self.idx_to_build_id[b_idx] for b_idx in shared_build_indices}
        for r in records:
            if r.build_id in target_build_set and r.builder_id != target_builder_id:
                c_idx = self.builder_id_to_idx.get(r.builder_id)
                if c_idx in builder_shared_stats:
                    builder_shared_stats[c_idx]["shared_builds"].add(r.build_id)
                    builder_shared_stats[c_idx]["shared_pieces"] += r.pieces_placed

        scores = np.zeros(len(candidate_co_builders), dtype=np.float64)

        for i, c_idx in enumerate(candidate_co_builders):
            c_row = sub_matrix.getrow(c_idx).toarray().flatten()  # shares of builder B
            min_shares = np.minimum(target_shares, c_row)
            weighted_overlap = min_shares * weights_sub
            
            # Kinship formula sum
            total_kinship = float(np.sum(weighted_overlap))
            scores[i] = total_kinship

            # Find key shared monument (highest single overlap build)
            if len(weighted_overlap) > 0:
                best_sub_idx = int(np.argmax(weighted_overlap))
                best_build_idx = shared_build_indices[best_sub_idx]
                builder_shared_stats[c_idx]["best_build_id"] = self.idx_to_build_id[best_build_idx]
                builder_shared_stats[c_idx]["best_build_score"] = float(weighted_overlap[best_sub_idx])

        # Rank candidates descending by KinshipScore
        ranked_indices = np.argsort(-scores)[:top_k]

        results: List[KinshipCollaborator] = []
        for rank_pos in ranked_indices:
            score = float(scores[rank_pos])
            if score <= 0.001:
                continue

            c_idx = candidate_co_builders[rank_pos]
            c_id = self.idx_to_builder_id[c_idx]
            c_name = self.builder_names.get(c_id, c_id)
            stats = builder_shared_stats[c_idx]

            # Derive role tag based on kinship magnitude & shared build ratio
            role_tag = self._derive_kinship_role_tag(
                kinship_score=score,
                shared_builds_count=len(stats["shared_builds"]),
                shared_pieces=stats["shared_pieces"]
            )

            results.append(
                KinshipCollaborator(
                    collaborator_id=c_id,
                    collaborator_name=c_name,
                    role_tag=role_tag,
                    shared_builds_count=len(stats["shared_builds"]),
                    shared_pieces_count=stats["shared_pieces"],
                    kinship_score=round(score * 10.0, 2),  # Scaled to intuitive 0-100 codex score
                    key_shared_build_id=stats["best_build_id"] or "shared_monument",
                )
            )

        return results

    @staticmethod
    def _derive_kinship_role_tag(
        kinship_score: float, shared_builds_count: int, shared_pieces: int
    ) -> str:
        """
        Norse codex heuristic mapping shared construction dynamics into role tags.
        """
        if shared_pieces > 10000 and shared_builds_count >= 10:
            return "Co-Architect (Shield-Wall)"
        elif shared_pieces > 5000:
            return "Forge & Mason Lead"
        elif shared_builds_count >= 8:
            return "Homestead Partner"
        elif shared_pieces > 2000:
            return "Timber Carpenter"
        elif shared_builds_count >= 4:
            return "Rampart Engineer"
        elif kinship_score > 2.5:
            return "Master Mason"
        else:
            return "Expedition Ally"

@router.get(
    "/{builder_id}/top8",
    response_model=List[KinshipCollaborator],
    summary="Compute Viking Shield-Wall Top 8 via NumPy Kinship Matrix",
    description=(
        "Dynamically calculates the Kinship Overlap score between the target builder "
        "and all co-builders using vectorized sparse matrices and logarithmic piece scaling."
    )
)
async def get_builder_shield_wall_top8(
    builder_id: str,
    top_k: int = Query(8, ge=1, le=16, description="Number of shield-wall allies to return"),
    engine_data=Depends(get_kinship_engine)
) -> List[KinshipCollaborator]:
    """
    FastAPI Route Handler for the dynamic Viking Shield-Wall Top 8.
    """
    engine, matrix, weights, records = engine_data
    
    top8 = engine.compute_kinship_for_builder(
        target_builder_id=builder_id,
        contribution_matrix=matrix,
        build_weights=weights,
        records=records,
        top_k=top_k
    )
    
    if not top8 and builder_id not in engine.builder_id_to_idx:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Builder '{builder_id}' has no historical piece attributions in the Comfy Archive."
        )
        
    return top8
